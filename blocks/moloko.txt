<script>
/* Файл забираем в фоне и отдаём браузеру готовым куском: атрибут
   download на чужом домене браузеры не слушают и вместо сохранения
   открывают PDF, показывая адрес хранилища. Так человек остаётся на
   нашей странице, а файл просто падает в «Загрузки». */
document.querySelectorAll('.km-pdf').forEach(function (a) {
  a.addEventListener('click', function (e) {
    e.preventDefault();
    if (a.dataset.busy) return;
    a.dataset.busy = '1';
    var было = a.textContent;
    a.textContent = 'Готовим файл…';
    fetch(a.href).then(function (r) { return r.blob(); }).then(function (b) {
      var u = URL.createObjectURL(b), l = document.createElement('a');
      l.href = u; l.download = a.dataset.imya;
      document.body.appendChild(l); l.click(); l.remove();
      setTimeout(function () { URL.revokeObjectURL(u); }, 10000);
      a.textContent = было; delete a.dataset.busy;
    }).catch(function () {
      a.textContent = было; delete a.dataset.busy;
      window.open(a.href, '_blank');
    });
  });
});
</script>
